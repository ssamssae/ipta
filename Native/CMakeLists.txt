cmake_minimum_required(VERSION 3.16)
project(ipta_transcriber LANGUAGES C CXX)
set(WHISPER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Vendor/whisper.cpp" CACHE PATH "whisper.cpp source")
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper)
add_executable(ipta-transcriber transcriber.cpp)
target_compile_features(ipta-transcriber PRIVATE cxx_std_17)
target_include_directories(ipta-transcriber PRIVATE "${WHISPER_SOURCE_DIR}/examples")
target_link_libraries(ipta-transcriber PRIVATE common whisper)
